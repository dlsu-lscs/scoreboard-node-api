```javascript
import express, { urlencoded, json } from "express";
import { initDB, closeDB } from "./config/connect.js";
import { auth } from "./config/auth.js";
import scoresRouter from "./routes/scores.routes.js";
import "dotenv/config";

const app = express();

const startServer = async () => {
  try {
    await initDB();
    console.log("Databases initialized.");

    // Parse JSON and URL-encoded bodies
    app.use(json());
    app.use(urlencoded({ extended: true }));

    // Mount better-auth BEFORE protected routes
    // This provides: /api/auth/sign-in/social, /api/auth/callback/google, /api/auth/sign-out, etc.
    app.all("/api/auth/*", auth.handler);

    // Mount protected score routes (requires API Key + JWT via middleware)
    app.use("/api/scores", scoresRouter);

    // Health check endpoint
    app.get("/", (req, res) => {
      res.status(200).json({
        status: "Ok",
        message: "LSCS Scoreboard API app is running.",
      });
    });

    const server = app.listen(process.env.PORT || 3000, () => {
      console.log(`Server running on port: ${process.env.PORT || 3000}`);
    });

    const shutdown = async () => {
      console.log("Shutting down...");
      server.close();
      try {
        await closeDB();
        console.log("All connections closed.");
        process.exit(0);
      } catch (err) {
        console.log("Error on shutdown: ", err);
        process.exit(1);
      }
    };

    process.on("SIGTERM", shutdown);
    process.on("SIGINT", shutdown);
  } catch (err) {
    console.error("Server startup error: ", err);
    process.exit(1);
  }
};

startServer().catch((err) => {
  console.error("Error during server initialization: ", err);
  process.exit(1);
});

export default app;
```

```javascript
import "dotenv/config";
import { auth } from "../config/auth.js";

/**
 * Validates API key from X-API-Key header
 * Per D-07: Migrated from Authorization: Bearer to X-API-Key header
 */
export function validateApiKey(req, res, next) {
  const apiKey = req.headers["x-api-key"];

  if (!apiKey) {
    return res.status(401).json({ message: "Unauthorized: No API key provided." });
  }

  if (apiKey !== process.env.API_SECRET) {
    return res.status(401).json({ message: "Unauthorized: Invalid API key." });
  }

  next();
}

/**
 * Validates session using better-auth
 * Per D-08: Composable middleware layer
 * Per D-03: JWT comes from Authorization header (frontend localStorage)
 */
export async function validateSession(req, res, next) {
  const authHeader = req.headers["authorization"];

  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return res.status(401).json({ message: "Unauthorized: No session token provided." });
  }

  const token = authHeader.split(" ")[1];

  try {
    const session = await auth.api.getSession({
      headers: new Headers({ authorization: `Bearer ${token}` }),
    });

    if (!session) {
      return res.status(401).json({ message: "Unauthorized: Invalid or expired session." });
    }

    // Attach user and session to request for downstream use
    req.user = session.user;
    req.session = session.session;
    next();
  } catch (error) {
    console.error("Session validation error:", error);
    return res.status(401).json({ message: "Unauthorized: Session validation failed." });
  }
}

// Legacy export for backward compatibility during transition
export { validateApiKey as authenticateApiSecret };
```

Do not introduce cookie-dependent session extraction in this phase; the protected API auth path is Bearer JWT per D-03.
