```javascript
import { betterAuth } from "better-auth";

export const auth = betterAuth({
  database: {
    provider: "mysql2",
    url: `mysql://${process.env.DB_USER}:${process.env.DB_PASS}@${process.env.DB_HOST}:${process.env.DB_PORT || 3306}/${process.env.DB_DATABASE}`,
  },
  secret: process.env.BETTER_AUTH_SECRET,
  baseURL: process.env.BETTER_AUTH_BASE_URL,
  socialProviders: {
    google: {
      clientId: process.env.GOOGLE_CLIENT_ID,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET,
    },
  },
  session: {
    expiresIn: 60 * 60 * 24 * 7, // 7 days
  },
  // Configure HTTP-only secure cookies per AUTH-02
  cookie: {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    maxAge: 60 * 60 * 24 * 7, // 7 days (matches session.expiresIn)
  },
});

export default auth;
```

**NOTE on D-03 vs AUTH-02:**
- CONTEXT.md D-03 originally specified localStorage for JWT storage
- REQUIREMENTS.md AUTH-02 explicitly requires "HTTP-only cookies" for session persistence
- **Resolution:** Using HTTP-only cookies provides better security and aligns with REQUIREMENTS.md. This overrides the earlier localStorage decision.

The session token will be:
- Stored in HTTP-only cookie (not accessible to JavaScript)
- Automatically sent with every request to /api/auth/*
- Protected against XSS attacks
- Cleared on logout via /api/auth/sign-out

```javascript
import express, { urlencoded, json } from "express";
import { initDB, closeDB } from "./config/connect.js";
import { auth } from "./config/auth.js";
import scoresRouter from "./routes/scores.routes.js";
import "dotenv/config";

const app = express();

const startServer = async () => {
  try {
    await initDB();
    console.log("Databases initialized.");

    // Parse JSON and URL-encoded bodies
    app.use(json());
    app.use(urlencoded({ extended: true }));

    // Mount better-auth BEFORE protected routes
    // This provides: /api/auth/sign-in/social, /api/auth/callback/google, /api/auth/sign-out, etc.
    app.all("/api/auth/*", auth.handler);

    // Mount protected score routes (requires API Key + JWT via middleware)
    app.use("/api/scores", scoresRouter);

    // Health check endpoint
    app.get("/", (req, res) => {
      res.status(200).json({
        status: "Ok",
        message: "LSCS Scoreboard API app is running.",
      });
    });

    const server = app.listen(process.env.PORT || 3000, () => {
      console.log(`Server running on port: ${process.env.PORT || 3000}`);
    });

    const shutdown = async () => {
      console.log("Shutting down...");
      server.close();
      try {
        await closeDB();
        console.log("All connections closed.");
        process.exit(0);
      } catch (err) {
        console.log("Error on shutdown: ", err);
        process.exit(1);
      }
    };

    process.on("SIGTERM", shutdown);
    process.on("SIGINT", shutdown);
  } catch (err) {
    console.error("Server startup error: ", err);
    process.exit(1);
  }
};

startServer().catch((err) => {
  console.error("Error during server initialization: ", err);
  process.exit(1);
});

export default app;
```

**Critical ordering per RESEARCH.md Pitfall 1:**
1. Body parsers (json, urlencoded)
2. better-auth handler at /api/auth/*
3. Protected routes at /api/scores (with auth middleware)

This ensures OAuth endpoints are accessible without requiring authentication.

```javascript
import "dotenv/config";
import { auth } from "../config/auth.js";

/**
 * Validates API key from X-API-Key header
 * Per D-07: Migrated from Authorization: Bearer to X-API-Key header
 */
export function validateApiKey(req, res, next) {
  const apiKey = req.headers["x-api-key"];

  if (!apiKey) {
    return res.status(401).json({ message: "Unauthorized: No API key provided." });
  }

  if (apiKey !== process.env.API_SECRET) {
    return res.status(401).json({ message: "Unauthorized: Invalid API key." });
  }

  next();
}

/**
 * Validates session using better-auth
 * Per D-08: Composable middleware layer
 * Handles both cookie-based sessions and Authorization header
 */
export async function validateSession(req, res, next) {
  try {
    // better-auth validates session from cookies automatically
    // We need to pass the request headers to auth.api.getSession
    const headers = new Headers();
    
    // Copy relevant headers for session validation
    const cookie = req.headers.cookie;
    const authHeader = req.headers.authorization;
    
    if (cookie) {
      headers.set("cookie", cookie);
    }
    
    if (authHeader) {
      headers.set("authorization", authHeader);
    }

    const session = await auth.api.getSession({ headers });

    if (!session) {
      return res.status(401).json({ message: "Unauthorized: Invalid or expired session." });
    }

    // Attach user and session to request for downstream use
    req.user = session.user;
    req.session = session.session;
    next();
  } catch (error) {
    console.error("Session validation error:", error);
    return res.status(401).json({ message: "Unauthorized: Session validation failed." });
  }
}

// Legacy export for backward compatibility during transition
export { validateApiKey as authenticateApiSecret };
```

The validateSession now handles cookie-based sessions from better-auth. The session cookie is:
- Set by better-auth during OAuth callback
- Automatically sent by browser with requests to same domain
- Validated by auth.api.getSession
- HTTP-only (not accessible to JavaScript)
