```javascript
import "dotenv/config";
import { auth } from "../config/auth.js";

/**
 * Validates API key from X-API-Key header
 * Per D-07: Migrated from Authorization: Bearer to X-API-Key header
 */
export function validateApiKey(req, res, next) {
  const apiKey = req.headers["x-api-key"];

  if (!apiKey) {
    return res.status(401).json({ message: "Unauthorized: No API key provided." });
  }

  if (apiKey !== process.env.API_SECRET) {
    return res.status(401).json({ message: "Unauthorized: Invalid API key." });
  }

  next();
}

/**
 * Validates JWT session using better-auth
 * Per D-08: Composable middleware layer
 * Per D-03: JWT stored in Authorization: Bearer header
 */
export async function validateSession(req, res, next) {
  const authHeader = req.headers["authorization"];

  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return res.status(401).json({ message: "Unauthorized: No session token provided." });
  }

  const token = authHeader.split(" ")[1];

  try {
    // Validate session with better-auth
    const session = await auth.api.getSession({
      headers: new Headers({ authorization: `Bearer ${token}` }),
    });

    if (!session) {
      return res.status(401).json({ message: "Unauthorized: Invalid or expired session." });
    }

    // Attach user to request for downstream use
    req.user = session.user;
    next();
  } catch (error) {
    console.error("Session validation error:", error);
    return res.status(401).json({ message: "Unauthorized: Session validation failed." });
  }
}

// Legacy export for backward compatibility during transition
export { validateApiKey as authenticateApiSecret };
```

Key changes per CONTEXT.md:
- D-07: Changed from `Authorization: Bearer <api-key>` to `X-API-Key: <api-key>`
- D-08: Composable middleware functions, each with single responsibility
- D-10: Returns 401 for auth failures with clear error messages
- Attach req.user from validated session for downstream use

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → API Key validation | Untrusted input (X-API-Key header) crosses boundary |
| API Key → Session validation | Only validated requests reach session layer |
| Session → Route handler | Only authenticated users reach business logic |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Spoofing | X-API-Key header | mitigate | validateApiKey compares against process.env.API_SECRET |
| T-02-02 | Tampering | JWT token | mitigate | validateSession uses auth.api.getSession which verifies signature |
| T-02-03 | Information Disclosure | Error messages | mitigate | Generic "Unauthorized" messages, log details server-side only |
| T-02-04 | Denial of Service | Session validation | accept | Rate limiting deferred to Phase 3 per CONTEXT.md |
| T-02-05 | Elevation of Privilege | req.user attachment | mitigate | req.user comes from validated session only, not client input |
</threat_model>
